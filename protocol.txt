# Get state of a board

Request:
GET /api/game/{gameId}/status

Response:
200
{
    "boardSize": 19,
    "boardState": [ # 0 -> empty, 1 -> black, 2 -> white
        [0, 1, 1, 0...],
        [0, 1, 0, 0...],
        ...
    ],
    "turn": 1, # even -> white's turn, odd -> black's turn
    "capturedBlack": 5,
    "capturedWhite": 3,
    "status": "PLAYING",
    "winnerId": 1
}

# Make a move

Request:
POST /api/game/{gameId}/move
{
    "playerId": 1, # 1 -> black, 2 -> white
    "moveType": "PLACE" # or "PASS" or "SURRENDER"
    "x": 10, # or -1 when not placing
    "y": 12
}

Response:
200
(no data)
or
400
{
    "message": "Invalid move: Pole jest już zajęte"
}
or
{
    "message": "Invalid move: Teraz jest ruch nie tego gracza"
}

# Negotiate scoring

Request:
POST /api/game/{gameId}/negotiate
{
    "playerId": 1, # 1 -> black, 2 -> white
    "livingWhite": 0,
    "deadWhite": 0,
    "whiteTerritory": 0,
    "livingBlack": 0,
    "deadBlack": 0,
    "blackTerritory": 0
}

Response:
200
(no data)
or
400
{
    "message": "Invalid move: Gra nie jest w fazie dogadywania"
}

# Get enemy's negotiation offer

Request:
POST /api/game/{gameId}/get_negotiation_details
{
    "playerId": 1 # 1 -> Black is asking - get white's offer, 2 -> the opposite
}

Response:
200
{
    "livingWhite": 0,
    "deadWhite": 0,
    "whiteTerritory": 0,
    "livingBlack": 0,
    "deadBlack": 0,
    "blackTerritory": 0
}

# Resume game

Request:
POST /api/game/{gameId}/resume
(no data)

Response:
200
(no data)
or
400
{
    "message": "Invalid request: Gra nie jest w fazie dogadywania"
}

# Create new game

Request:
POST /api/new_game
{
    "boardSize": 19,
    "requestedColor": "BLACK"
}

Response:
200
{
    "gameId": " ABC-123"
    "playerId": 1 # 1 -> black, 2 -> white
}

# Join game

Request:
POST /api/game/{gameId}/join
(no data)

Response:
200
{
    "playerId": 2 # 1 -> black, 2 -> white
}
or
400
{
    "message": " Can't join: Gra już się rozpoczęła"
